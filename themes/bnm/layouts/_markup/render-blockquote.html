{{- /* Callouts: Hugo alert syntax (> [!NOTE] Optional title). Built-in types plus update and correction, which Hugo passes through as AlertType like any other word. Anything else, and plain blockquotes, render as an ordinary
. The label is capitalised here, not by CSS, and is real text so screen readers announce it. Margin variant: > [!NOTE margin] Optional title. Hugo's alert parser only accepts [!LETTERS], so this arrives as a regular blockquote and the first line of .Text is parsed here. Only note and tip get the callout-margin class (js/footnotes.js moves it into the right margin on wide screens); the marker is accepted but ignored on the other types. Markup is otherwise identical, so without JS or below the breakpoint it is an ordinary in-column callout. */ -}} {{- $labels := dict "note" "Note" "tip" "Tip" "important" "Important" "warning" "Warning" "caution" "Caution" "update" "Update" "correction" "Correction" -}} {{- $type := "" -}} {{- $title := "" -}} {{- $text := .Text -}} {{- $margin := false -}} {{- if eq .Type "alert" -}} {{- $type = .AlertType -}} {{- $title = .AlertTitle -}} {{- else -}} {{- with findRESubmatch `(?i)^

\[!(note|tip|important|warning|caution|update|correction) margin\][ \t]*(.*?)(?:

|\n)` (string .Text) 1 -}} {{- $m := index . 0 -}} {{- $type = lower (index $m 1) -}} {{- $title = index $m 2 -}} {{- $rest := strings.TrimPrefix (index $m 0) (string $.Text) -}} {{- if strings.HasSuffix (index $m 0) "\n" }}{{ $rest = printf "

%s" $rest }}{{ end -}} {{- $text = $rest -}} {{- $margin = in (slice "note" "tip") $type -}} {{- end -}} {{- end -}} {{- if and $type (isset $labels $type) -}}

{{ index $labels $type }}{{ with $title }} · {{ . | safeHTML }}{{ end }}

{{ $text | safeHTML }}
{{- else -}}
{{ .Text }}
{{- end -}}
