{{/* Returns CSS: @font-face rules plus --font- custom properties, generated from [params.fonts.] (family, fallback, dir, files[{file, weight, style, preload}]). The body font may also carry its own type settings, because the same size looks different in different faces: size -> --text-base, leading -> --leading, measure -> --measure. Files live in static/fonts// (theme or site). */}} {{- $faces := "" -}} {{- $vars := "" -}} {{- range $role, $f := site.Params.fonts -}} {{- range $f.files -}} {{- $url := printf "fonts/%s/%s" $f.dir .file | relURL -}} {{- $faces = printf `%s@font-face{font-family:%q;src:url(%q) format("woff2");font-weight:%s;font-style:%s;font-display:swap}%s` $faces $f.family $url (.weight | default "400") (.style | default "normal") "\n" -}} {{- end -}} {{- $vars = printf "%s--font-%s:%q,%s;" $vars $role $f.family $f.fallback -}} {{- if eq $role "body" -}} {{- with $f.size }}{{ $vars = printf "%s--text-base:%s;" $vars . }}{{ end -}} {{- with $f.leading }}{{ $vars = printf "%s--leading:%v;" $vars . }}{{ end -}} {{- with $f.measure }}{{ $vars = printf "%s--measure:%s;" $vars . }}{{ end -}} {{- end -}} {{- end -}} {{- return (printf "%s:root{%s}\n" $faces $vars) -}}
